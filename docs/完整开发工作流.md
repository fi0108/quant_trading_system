```python
def process(self, data: Dict[str, Any]) -> Dict[str, Any]:
    """处理数据"""
    # 1. 验证输入
    if not data:
        raise ValueError("Data cannot be empty")
    
    # 2. 提取字段
    symbol = data.get('symbol')
    price = data.get('price')
    
    if not symbol or price is None:
        raise ValueError("Missing required fields")
    
    # 3. 处理逻辑
    result = self._process_internal(symbol, price)
    
    # 4. 记录日志
    logger.info(f"Processed {symbol} at ${price}")
    
    return result

def _process_internal(self, symbol: str, price: float) -> Dict[str, Any]:
    """内部处理逻辑"""
    # 实现细节
    return {'status': 'success', 'symbol': symbol, 'price': price}
```

```python
"""
功能模块单元测试
"""

import pytest
from unittest.mock import Mock, patch

from module_name.feature import FeatureManager


class TestFeatureManager:
    """FeatureManager 测试类"""
    
    def setup_method(self):
        """每个测试前的设置"""
        self.config = {'key': 'value'}
        self.manager = FeatureManager(self.config)
    
    def test_initialization(self):
        """测试：初始化"""
        assert self.manager.config == self.config
    
    def test_process_success(self):
        """测试：正常处理"""
        data = {'symbol': 'AAPL', 'price': 150.0}
        result = self.manager.process(data)
        
        assert result['status'] == 'success'
        assert result['symbol'] == 'AAPL'
    
    def test_process_missing_data(self):
        """测试：数据缺失"""
        with pytest.raises(ValueError, match="Missing required fields"):
            self.manager.process({'symbol': 'AAPL'})
    
    def test_process_empty_data(self):
        """测试：空数据"""
        with pytest.raises(ValueError, match="Data cannot be empty"):
            self.manager.process({})
    
    @patch('module_name.feature.logger')
    def test_logging(self, mock_logger):
        """测试：日志记录"""
        data = {'symbol': 'AAPL', 'price': 150.0}
        self.manager.process(data)
        
        mock_logger.info.assert_called_once()
```

这个时间远远小于重复开发和返工的时间（几小时）。

---

**Q3: 每个步骤都要做吗？**

A: 核心步骤必须做：
- ✅ 必须：开发前查模块索引
- ✅ 必须：开发前检查清单
- ✅ 必须：单元测试通过
- ✅ 必须：端到端验证
- ✅ 必须：更新模块索引
- ✅ 必须：更新架构演进日志

可选步骤：
- ⭕ 可选：周度总结（小项目可以月度）
- ⭕ 可选：完成总结（Task很小可以简化）
